    \begin{problema}
        Sea $x$ un n mero real. Prueba, usando la unicidad del inverso aditivo,
        lo siguiente:
        \begin{multicols}{2}
            \begin{enumerate}
                \item[(a)] $\left(-1\right)x=-x.$
                \item[(b)] $-\left(-x\right)=x.$
            \end{enumerate}
        \end{multicols}
        \textbf{Sugerencia:} Para proba, que por ejemplo, el inciso (a) muestra primero que $(-1)x+x=0$ y posteriormente usa la unicidad del invesro aditivo.
    \end{problema}

    \begin{problema}
        Sea $x$ un número real distinto de cero. Prueba, usando la unicidad del inverso multiplicativo, que
        \begin{equation*}
            (x^{-1})^{-1}=x.
        \end{equation*}
    \end{problema}

    \begin{problema}
        Sea $A$ un conjunto no vacío. Prueba que si $x\lt c$ para toda $x\in A$, entonces
        \begin{equation*}
            \sup(A)\leq c.
        \end{equation*}
    \end{problema}
