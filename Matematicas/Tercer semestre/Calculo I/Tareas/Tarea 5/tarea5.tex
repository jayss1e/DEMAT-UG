    \begin{problema}
        Prueba que si $x$
        y $y$ son números reales, con $y\neq0,$ entonces 
        \begin{enumerate}
            \item[(a)] $\frac{+x}{+y}=+\frac{x}{y}.$ \vspace{0.1cm}
            \item[(b)] $\frac{-x}{+y}=-\frac{x}{y}.$ \vspace{0.1cm}
            \item[(c)] $\frac{+x}{-y}=-\frac{x}{y}.$ \vspace{0.1cm}
            \item[(d)] $\frac{-x}{-y}=+\frac{x}{y}.$ 
        \end{enumerate}
        Las leyes de los signos para la división se resumen de manera conveniente
        como 
        \begin{equation*}
            \begin{array}{ccccc}
                + & \text{entre} & + & \text{da} & +\\
                - & \text{entre} & + & \text{da} & -\\
                + & \text{entre} & - & \text{da} & -\\
                - & \text{entre} & - & \text{da} & +
            \end{array}
        \end{equation*}
        \textbf{Sugerencia 1:} El inciso (a) es claro, pues $+x=x,$ $+y=y$
        y $+\left(x/y\right)=x/y.$ Usa la unicidad del inverso aditivo para
        probar los incisos (b) y (c). Finalmente, deduce (d) de los incisos
        (b) y (c). \textbf{Sugerencia 2:} Para probar (b), usa las leyes de
        los signos para la multiplicación, y para probar (c), usa que $\left(-y\right)^{-1}=-\left(y^{-1}\right)$
        y las leyes de los signos para la multiplicación.
    \end{problema}

    \begin{proof}
        \textbf{(a)} Es claro, pues $+x=x$, $+y=y$ y
        \begin{equation*}
            +\left(\frac{x}{y}\right)=\frac{x}{y}.
        \end{equation*}
        Recordemos que, por definición,
        \begin{equation*}
            \frac{a}{b}=a\cdot b^{-1}
        \end{equation*}
        para $a,b\in\mathbb{R}$ con $b\neq0$. Además, para todo $a\in\mathbb{R}$ se tiene que $(-1)\cdot a=-a$. En efecto,
        \begin{equation*}
            a+(-1)\cdot a=1\cdot a+(-1)\cdot a=(1+(-1))\cdot a=0\cdot a=0,
        \end{equation*}
        así que $(-1)\cdot a$ es un inverso aditivo de $a$ y, por la unicidad del inverso aditivo, $(-1)\cdot a=-a$.

        \textbf{(b)} Por la definición de división,
        \begin{equation*}
            \frac{-x}{y}=(-x)\cdot y^{-1}.
        \end{equation*}
        Escribimos $-x$ como $(-1)\cdot x$. Así,
        \begin{equation*}
            (-x)\cdot y^{-1}=((-1)\cdot x)\cdot y^{-1}.
        \end{equation*}
        Por la asociatividad de la multiplicación,
        \begin{equation*}
            ((-1)\cdot x)\cdot y^{-1}=(-1)\cdot(x\cdot y^{-1}).
        \end{equation*}
        Aplicando de nuevo que $(-1)\cdot a=-a$, ahora con $a=x\cdot y^{-1}$, tenemos que
        \begin{equation*}
            (-1)\cdot(x\cdot y^{-1})=-(x\cdot y^{-1})=-\frac{x}{y}.
        \end{equation*}
        Por lo tanto,
        \begin{equation*}
            \frac{-x}{y}=-\frac{x}{y}.
        \end{equation*}

        \textbf{(c)} Como $y\neq0$, también $-y\neq0$, así que $(-y)^{-1}$ existe. Usando la definición de división,
        \begin{equation*}
            \frac{x}{-y}=x\cdot(-y)^{-1}.
        \end{equation*}
        Por las leyes de los signos para la multiplicación, $(-y)\cdot(-y^{-1})=y\cdot y^{-1}=1$, es decir, $-y^{-1}$ es un inverso multiplicativo de $-y$. Por la unicidad del inverso multiplicativo, $(-y)^{-1}=-y^{-1}$. De esto se sigue que
        \begin{equation*}
            \frac{x}{-y}=x\cdot(-y)^{-1}=x\cdot(-y^{-1})=-(x\cdot y^{-1})=-\frac{x}{y},
        \end{equation*}
        donde la penúltima igualdad se debe, de nuevo, a las leyes de los signos para la multiplicación. Por lo tanto,
        \begin{equation*}
            \frac{x}{-y}=-\frac{x}{y}.
        \end{equation*}

        \textbf{(d)} Aplicando la definición de división, lo probado en (c) y que $(-1)\cdot a=-a$, tenemos que
        \begin{equation*}
            \frac{-x}{-y}=(-x)\cdot(-y)^{-1}=(-x)\cdot(-y^{-1})=((-1)\cdot x)\cdot((-1)\cdot y^{-1}).
        \end{equation*}
        Por la asociatividad y la conmutatividad de la multiplicación,
        \begin{equation*}
            ((-1)\cdot x)\cdot((-1)\cdot y^{-1})=((-1)\cdot(-1))\cdot(x\cdot y^{-1}).
        \end{equation*}
        Por las leyes de los signos, $(-1)\cdot(-1)=1$, por lo que
        \begin{equation*}
            ((-1)\cdot(-1))\cdot(x\cdot y^{-1})=1\cdot(x\cdot y^{-1})=x\cdot y^{-1}=\frac{x}{y}.
        \end{equation*}
        Por lo tanto,
        \begin{equation*}
            \frac{-x}{-y}=\frac{x}{y}.
        \end{equation*}
    \end{proof}

    \begin{problema}
        Prueba que si $x<y+\e$ para todo $\e>0$, entonces $x\leq y$.
    \end{problema}

    \begin{proof}
        Procederemos por contradicción. Supongamos que $x\leq y$ es falso, es decir, que $x>y$. Entonces $x-y>0$, de modo que podemos tomar $\e=x-y$ en la hipótesis y obtenemos
        \begin{align*}
            x&<y+\e\\
            &=y+(x-y)\\
            &=x,
        \end{align*}
        es decir, $x<x$, lo cual es una contradicción. Por lo tanto, $x\leq y$.
    \end{proof}

    \begin{problema}
        Sean $A$ y $B$ subconjuntos no vacíos y acotados de números reales tales que $A\subset B$. Prueba que
        \begin{equation*}
            \inf(B)\leq\inf(A)\leq\sup(A)\leq\sup(B).
        \end{equation*}
    \end{problema}

    \begin{proof}
        Como $A$ y $B$ son no vacíos y acotados, por el axioma del supremo existen $\inf(B)$, $\inf(A)$, $\sup(A)$ y $\sup(B)$ en $\R$.

        Sea $a\in A$. Dado que $A\subset B$, se tiene que $a\in B$. Por definición de ínfimo de $B$, $\inf(B)$ es una cota inferior de $B$; en particular, $\inf(B)\leq a$. Como esto vale para todo $a\in A$, $\inf(B)$ es una cota inferior de $A$. Por definición del ínfimo como la mayor de las cotas inferiores, se sigue que
        \begin{equation*}
            \inf(B)\leq\inf(A).
        \end{equation*}
        Como $A$ es no vacío, existe algún $a\in A$. Por definición de ínfimo y supremo, $\inf(A)\leq a$ y $a\leq\sup(A)$, de modo que, por transitividad,
        \begin{equation*}
            \inf(A)\leq\sup(A).
        \end{equation*}
        Sea $a\in A$. Nuevamente, $a\in B$ porque $A\subset B$. Por definición de supremo de $B$, $\sup(B)$ es una cota superior de $B$, luego $a\leq\sup(B)$. Como esto vale para todo $a\in A$, $\sup(B)$ es una cota superior de $A$. Por definición del supremo como la menor de las cotas superiores, se sigue que
        \begin{equation*}
            \sup(A)\leq\sup(B).
        \end{equation*}
        Combinando las tres desigualdades obtenemos que
        \begin{equation*}
            \inf(B)\leq\inf(A)\leq\sup(A)\leq\sup(B).
        \end{equation*}
    \end{proof}

    \begin{problema}
        Dados $n$ y $k$ números enteros
        con $0\leq k\leq n$ definimos el \textit{coeficiente binomial} 
        \begin{equation*}
            \binom{n}{k}=\frac{n!}{k!(n-k)!}.
        \end{equation*}
        Recuerda que el número factorial de un número natural $n$ se define
        como $n!=n\cdot\left(n-1\right)\cdots2\cdot1$ y que $0!=1.$ 
        \begin{enumerate}
            \item[(a)]  Sea $n$ un número natural. Prueba que 
            \begin{equation*}
                \binom{n}{0}=\binom{n}{n}=1.
            \end{equation*}
            \item[(b)]  Sean $n$ y $k$ números naturales con $k\leq n.$ Prueba que 
            \begin{equation*}
                \binom{n}{k-1}+\binom{n}{k}=\binom{n+1}{k}.
            \end{equation*}
        \end{enumerate}
    \end{problema}

    \begin{proof}
        \textbf{(a)} Por definición, y usando que $0!=1$,
        \begin{equation*}
            \binom{n}{0}=\frac{n!}{0!\,n!}=\frac{n!}{n!}=1
        \end{equation*}
        y
        \begin{equation*}
            \binom{n}{n}=\frac{n!}{n!\,0!}=\frac{n!}{n!}=1.
        \end{equation*}

        \textbf{(b)} Ponemos ambos sumandos con un mismo denominador. Usando que $k!=k\cdot(k-1)!$, tenemos que
        \begin{equation*}
            \binom{n}{k-1}=\frac{n!}{(k-1)!\,(n-k+1)!}=\frac{n!\,k}{k!\,(n-k+1)!},
        \end{equation*}
        y usando que $(n-k+1)!=(n-k+1)\cdot(n-k)!$, tenemos que
        \begin{equation*}
            \binom{n}{k}=\frac{n!}{k!\,(n-k)!}=\frac{n!\,(n-k+1)}{k!\,(n-k+1)!}.
        \end{equation*}
        Sumando,
        \begin{equation*}
            \binom{n}{k-1}+\binom{n}{k}=\frac{n!}{k!\,(n-k+1)!}\bigl(k+(n-k+1)\bigr)=\frac{n!\,(n+1)}{k!\,(n-k+1)!}.
        \end{equation*}
        Como $(n+1)\cdot n!=(n+1)!$ y $n-k+1=(n+1)-k$, concluimos que
        \begin{equation*}
            \binom{n}{k-1}+\binom{n}{k}=\frac{(n+1)!}{k!\,((n+1)-k)!}=\binom{n+1}{k}.
        \end{equation*}
    \end{proof}

    \begin{problema}
        Prueba, mediante contraejemplos, que es falso que la suma, resta, producto y cociente de números irracionales sea un número irracional.
    \end{problema}

    \begin{proof}
        Basta exhibir, para cada operación, dos números irracionales cuyo resultado sea racional.

        \textbf{Suma.} Consideremos los números
        \begin{equation*}
            1+\sqrt{5}\qquad\text{y}\qquad1-\sqrt{5}.
        \end{equation*}
        Ambos son irracionales: si $1+\sqrt{5}=r$ con $r\in\mathbb{Q}$, entonces $\sqrt{5}=r-1\in\mathbb{Q}$, lo cual es falso, pues $\sqrt{5}$ es irracional; de manera análoga, $1-\sqrt{5}$ es irracional. Sin embargo, su suma es
        \begin{align*}
            (1+\sqrt{5})+(1-\sqrt{5})&=(1+1)+(\sqrt{5}-\sqrt{5})\\
            &=2+0\\
            &=2,
        \end{align*}
        que es racional. Así, la suma de irracionales no es necesariamente irracional.

        \textbf{Resta, producto y cociente.} Consideremos el número irracional $\sqrt{2}$. Su diferencia consigo mismo es
        \begin{equation*}
            \sqrt{2}-\sqrt{2}=0,
        \end{equation*}
        su producto por sí mismo es
        \begin{equation*}
            \sqrt{2}\cdot\sqrt{2}=2,
        \end{equation*}
        y su cociente entre sí mismo es
        \begin{equation*}
            \frac{\sqrt{2}}{\sqrt{2}}=1.
        \end{equation*}
        Los tres resultados son racionales. Por lo tanto, la resta, el producto y el cociente de irracionales tampoco son necesariamente irracionales.
    \end{proof}

    \begin{problema}
        Sean $f:A\rightarrow B$ y $C\subset B.$ Definimos la \textit{imagen
        inversa} \index{imagen inversa de un conjunto bajo una función} del
        conjunto $C$ bajo la función $f$ como el conjunto 
        \begin{equation*}
            f^{-1}\left(C\right)=\left\{ x\in A:f\left(x\right)\in C\right\} .
        \end{equation*}
        Dados $C,D\subset B,$ prueba lo siguiente:
        \begin{enumerate}
            \item[(a)] $f^{-1}\left(C\cup D\right)=f^{-1}\left(C\right)\cup f^{-1}\left(D\right).$
            \item[(b)] $f^{-1}\left(C\cap D\right)=f^{-1}\left(C\right)\cap f^{-1}\left(D\right).$ 
            \item[(c)] $f^{-1}\left(C\backslash D\right)=f^{-1}\left(C\right)\backslash f^{-1}\left(D\right).$
        \end{enumerate}
    \end{problema}

    \begin{proof}
        En los tres incisos probaremos la igualdad de conjuntos por doble contención, usando que, para todo $x\in A$ y todo $E\subset B$,
        \begin{equation*}
            x\in f^{-1}(E)\quad\text{si y sólo si}\quad f(x)\in E.
        \end{equation*}

        \textbf{(a)} ($\subseteq$) Sea $x\in f^{-1}(C\cup D)$. Entonces $f(x)\in C\cup D$, es decir, $f(x)\in C$ o $f(x)\in D$. En el primer caso, $x\in f^{-1}(C)$; en el segundo, $x\in f^{-1}(D)$. Por lo tanto, $x\in f^{-1}(C)\cup f^{-1}(D)$.

        ($\supseteq$) Sea $x\in f^{-1}(C)\cup f^{-1}(D)$. Entonces $x\in f^{-1}(C)$ o $x\in f^{-1}(D)$. Si $x\in f^{-1}(C)$, entonces $f(x)\in C\subseteq C\cup D$. Si $x\in f^{-1}(D)$, entonces $f(x)\in D\subseteq C\cup D$. En cualquier caso, $f(x)\in C\cup D$, es decir, $x\in f^{-1}(C\cup D)$.

        Concluimos que
        \begin{equation*}
            f^{-1}(C\cup D)=f^{-1}(C)\cup f^{-1}(D).
        \end{equation*}

        \textbf{(b)} ($\subseteq$) Sea $x\in f^{-1}(C\cap D)$. Entonces $f(x)\in C\cap D$, es decir, $f(x)\in C$ y $f(x)\in D$. Por lo tanto, $x\in f^{-1}(C)$ y $x\in f^{-1}(D)$, luego $x\in f^{-1}(C)\cap f^{-1}(D)$.

        ($\supseteq$) Sea $x\in f^{-1}(C)\cap f^{-1}(D)$. Entonces $x\in f^{-1}(C)$ y $x\in f^{-1}(D)$, lo que equivale a que $f(x)\in C$ y $f(x)\in D$. Por lo tanto, $f(x)\in C\cap D$, es decir, $x\in f^{-1}(C\cap D)$.

        Concluimos que
        \begin{equation*}
            f^{-1}(C\cap D)=f^{-1}(C)\cap f^{-1}(D).
        \end{equation*}

        \textbf{(c)} ($\subseteq$) Sea $x\in f^{-1}(C\backslash D)$. Entonces $f(x)\in C\backslash D$, es decir, $f(x)\in C$ y $f(x)\notin D$. Por lo tanto, $x\in f^{-1}(C)$ y $x\notin f^{-1}(D)$, luego $x\in f^{-1}(C)\backslash f^{-1}(D)$.

        ($\supseteq$) Sea $x\in f^{-1}(C)\backslash f^{-1}(D)$. Entonces $x\in f^{-1}(C)$ y $x\notin f^{-1}(D)$, es decir, $f(x)\in C$ y $f(x)\notin D$. Por definición de diferencia de conjuntos, $f(x)\in C\backslash D$, de modo que $x\in f^{-1}(C\backslash D)$.

        Concluimos que
        \begin{equation*}
            f^{-1}(C\backslash D)=f^{-1}(C)\backslash f^{-1}(D).
        \end{equation*}
    \end{proof}

    \begin{problema}
        Determina si las funciones siguientes son 1-1 o sobre (recuerda que
        si no se especifica el codominio de una función, se supone, por sencillez,
        que su codominio es todo $\mathbb{R}$).
        \begin{multicols}{2}
            \begin{enumerate}
                \item[(a)] $f\left(x\right)=x^{2}+1.$
                \item[(b)] $g\left(x\right)=\sqrt{x+1}+1.$
                \item[(c)] $h\left(x\right)=2x^{3}+x.$ 
            \end{enumerate}
        \end{multicols}
    \end{problema}

    \begin{proof}
        \textbf{(a)} $f$ no es 1-1, ya que
        \begin{equation*}
            f(1)=2=f(-1),
        \end{equation*}
        pero $1\neq-1$. Tampoco es sobre. Consideremos $y=0$; para que $f$ sea sobre, debe existir un $x\in\R$ tal que $f(x)=0$, es decir,
        \begin{equation*}
            x^{2}+1=0,\quad\text{o sea,}\quad x^{2}=-1.
        \end{equation*}
        Sin embargo, $x^{2}\geq0$ para todo $x\in\R$, así que no existe ningún $x\in\R$ cuyo cuadrado sea $-1$. Por lo tanto, $0$ no está en la imagen de $f$.

        \textbf{(b)} El dominio de $g$ es el conjunto de los $x\in\R$ tales que $x+1\geq0$, es decir, $[-1,\infty)$. La función $g$ es 1-1. En efecto, sean $x_{1},x_{2}\in[-1,\infty)$ tales que $g(x_{1})=g(x_{2})$. Entonces
        \begin{align*}
            \sqrt{x_{1}+1}+1&=\sqrt{x_{2}+1}+1,\\
            \sqrt{x_{1}+1}&=\sqrt{x_{2}+1}.
        \end{align*}
        Elevando al cuadrado y usando que $\bigl(\sqrt{t}\bigr)^{2}=t$ para todo $t\geq0$, obtenemos
        \begin{align*}
            x_{1}+1&=x_{2}+1,\\
            x_{1}&=x_{2}.
        \end{align*}
        Así, $g$ es 1-1. Sin embargo, $g$ no es sobre. En efecto, si tomamos $y=0$, para que $g$ sea sobre debe existir un $x\in[-1,\infty)$ tal que $g(x)=0$, es decir,
        \begin{equation*}
            \sqrt{x+1}+1=0,\quad\text{o sea,}\quad\sqrt{x+1}=-1,
        \end{equation*}
        lo cual no puede suceder, pues la raíz cuadrada de un número nunca es negativa. Así, $g$ no es sobre.

        \textbf{(c)} $h$ es 1-1. En efecto, sean $a,b\in\R$ tales que $h(a)=h(b)$. Entonces
        \begin{equation*}
            2a^{3}+a=2b^{3}+b,
        \end{equation*}
        por lo que
        \begin{equation*}
            2a^{3}-2b^{3}+a-b=0.
        \end{equation*}
        Por la diferencia de cubos, $a^{3}-b^{3}=(a-b)(a^{2}+ab+b^{2})$, así que
        \begin{equation*}
            2(a-b)(a^{2}+ab+b^{2})+(a-b)=0.
        \end{equation*}
        Factorizando $a-b$,
        \begin{equation*}
            (a-b)\bigl[2(a^{2}+ab+b^{2})+1\bigr]=0.
        \end{equation*}
        Completando el cuadrado,
        \begin{equation*}
            a^{2}+ab+b^{2}=\left(a+\frac{b}{2}\right)^{2}+\frac{3b^{2}}{4}\geq0,
        \end{equation*}
        de modo que $2(a^{2}+ab+b^{2})+1\geq1>0$. Como el producto es cero y el segundo factor no lo es, necesariamente $a-b=0$, es decir, $a=b$. Así, $h$ es 1-1.
    \end{proof}

    \begin{problema}
        A partir de la definición del valor absoluto de un número, prueba la proposición 31.
    \end{problema}

    \begin{proof}
        Recordemos que la proposición 31 afirma que, para todos $x,y\in\R$,
        \begin{enumerate}
            \item[(a)] $|x+y|\leq|x|+|y|$ (desigualdad del triángulo);
            \item[(b)] $|xy|=|x||y|$.
        \end{enumerate}
        Por definición,
        \begin{equation*}
            |t|=\begin{cases}
                t, & \text{si }t\geq 0,\\
                -t, & \text{si }t<0.
            \end{cases}
        \end{equation*}
        Notemos que si $t\leq0$, entonces $|t|=-t$: si $t<0$ es la definición, y si $t=0$, entonces $|t|=0=-t$.

        \textbf{(a)} Primero veamos que, para todo $t\in\R$, se cumple
        \begin{equation*}
            -|t|\leq t\leq|t|.
        \end{equation*}
        En efecto, si $t\geq0$, entonces $|t|=t$ y $-|t|=-t\leq0\leq t$. Si $t<0$, entonces $|t|=-t>0$, de modo que $t<0<|t|$ y $-|t|=t$.

        Aplicamos esta propiedad a $x$ y a $y$:
        \begin{equation*}
            -|x|\leq x\leq|x|,\qquad -|y|\leq y\leq|y|.
        \end{equation*}
        Sumando miembro a miembro las desigualdades, obtenemos
        \begin{equation*}
            -(|x|+|y|)\leq x+y\leq|x|+|y|.
        \end{equation*}
        Ahora usamos la definición de $|x+y|$. Si $x+y\geq0$, entonces, por la desigualdad de la derecha,
        \begin{equation*}
            |x+y|=x+y\leq|x|+|y|.
        \end{equation*}
        Si $x+y<0$, entonces, multiplicando por $-1$ la desigualdad de la izquierda,
        \begin{equation*}
            |x+y|=-(x+y)\leq|x|+|y|.
        \end{equation*}
        En ambos casos, $|x+y|\leq|x|+|y|$.

        \textbf{(b)} Consideramos los signos de $x$ y $y$. Si $x\geq0$ y $y\geq0$, entonces $xy\geq0$, así que
        \begin{equation*}
            |xy|=xy=|x||y|.
        \end{equation*}
        Si $x\geq0$ y $y<0$, entonces $xy\leq0$, así que
        \begin{equation*}
            |xy|=-(xy)=x(-y)=|x||y|.
        \end{equation*}
        Si $x<0$ y $y\geq0$, entonces $xy\leq0$, así que
        \begin{equation*}
            |xy|=-(xy)=(-x)y=|x||y|.
        \end{equation*}
        Si $x<0$ y $y<0$, entonces $xy>0$, así que
        \begin{equation*}
            |xy|=xy=(-x)(-y)=|x||y|.
        \end{equation*}
        En todos los casos, $|xy|=|x||y|$.
    \end{proof}

    \begin{problema}
        Prueba, usando la definición de límite, que
        \begin{equation*}
            \lim_{x\to1}x+2=3.
        \end{equation*}
    \end{problema}

    \begin{proof}
        Sea $\e>0$ arbitrario y tomemos $\delta=\e$, que es positivo. Si $x\in\R$ cumple $0<|x-1|<\delta$, entonces
        \begin{align*}
            |f(x)-3|&=|(x+2)-3|\\
            &=|x-1|\\
            &<\delta\\
            &=\e.
        \end{align*}
        Como $\e>0$ era arbitrario, concluimos que
        \begin{equation*}
            \lim_{x\to1}x+2=3.\qedhere
        \end{equation*}
    \end{proof}

    \begin{problema}
        Prueba, usando la negación de la definición de límite, que es \textbf{falso} que
        \begin{equation*}
            \lim_{x\to1}x+2=5.
        \end{equation*}
    \end{problema}

    \begin{proof}
        Negando la definición de límite, debemos probar que existe $\e>0$ tal que para todo $\delta>0$ existe $x\in\R$ con
        \begin{equation*}
            0<|x-1|<\delta\qquad\text{y}\qquad|f(x)-5|\geq\e.
        \end{equation*}

        Tomemos $\e=1$ y sea $\delta>0$ arbitrario. Consideremos
        \begin{equation*}
            x=1-\frac{\delta}{2}.
        \end{equation*}
        Entonces $|x-1|=\delta/2$, de modo que $0<|x-1|<\delta$. Además, como $\delta>0$, se tiene $x<1<3$, así que $x-3<0$ y
        \begin{align*}
            |f(x)-5|&=|x-3|\\
            &=3-x\\
            &=2+\frac{\delta}{2}\\
            &>1=\e.
        \end{align*}
        Como $\delta>0$ era arbitrario, se cumple la negación de la definición de límite. Por lo tanto, es falso que
        \begin{equation*}
            \lim_{x\to1}x+2=5.\qedhere
        \end{equation*}
    \end{proof}
