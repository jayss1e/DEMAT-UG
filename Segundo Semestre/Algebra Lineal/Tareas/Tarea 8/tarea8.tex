    \begin{mdframed}[style=mdbluebox,frametitle={Ejercicio 1}]
        Dado un espacio vectorial $V$ de dimensión finita $n$ sobre $F$ y dada una base ordenada $\mathcal{B}$ de $V$. Demuestra
        que para todo $v,w\in V$ y $a\in F$, se tiene que:
        \begin{equation*}
            [av+w]_{\mathcal{B}}=a[v]_{\mathcal{B}}+[w]_{\mathcal{B}}.
        \end{equation*}
    \end{mdframed}

    \begin{proof}
        Por definición de coordenadas respecto a la base $\BB$, para cualesquiera $v,w\in V$, existen
        escalares únicos $\alpha_{1},\dots,\alpha_{n},\beta_{1},\dots,\beta_{n}\in F$ tales que
        \begin{equation*}
            v=\sum_{i=1}^{n}\alpha_{i}v_{i},\qquad\sum_{i=1}^{n}\beta_{i}v_{i}.
        \end{equation*}
        Entonces,
        \begin{equation*}
            av+w=a\paren{\sum_{i=1}^{n}\alpha_{i}v_{i}}+\sum_{i=1}^{n}\beta_{i}v_{i}=\sum_{i=1}^{n}a\alpha_{i}v_{i}+\sum_{i=1}^{n}\beta_{i}v_{i}=\sum_{i=1}^{n}(a\alpha_{i}+\beta_{i})v_{i}.
        \end{equation*}
        Se sigue que
        \begin{equation*}
            [av+w]_{\BB}=(a\alpha_{1}+\beta_{1},\dots,a\alpha_{n}+\beta_{n}).
        \end{equation*}
        Por otro lado,
        \begin{equation*}
            [v]_{\BB}=(\alpha_{1},\dots,\alpha_{n}),\qquad [w]_{\BB}=(\beta_{1},\dots,\beta_{n}).
        \end{equation*}
        Entonces,
        \begin{equation*}
            a[v]_{\BB}+[w]_{\BB}=(a\alpha_{1},\dots,a\alpha_{n})+(\beta_{1},\dots,\beta_{n})=(a\alpha_{1}+\beta_{1},\dots,a\alpha_{n}+\beta_{n})
        \end{equation*}
        Por lo tanto,
        \begin{equation*}
            [av+w]_{\BB}=a[v]_{\BB}+[w]_{\BB}.
        \end{equation*}
    \end{proof}

    \begin{mdframed}[style=mdbluebox,frametitle={Ejercicio 2}]
        Demuestra que $\mathcal{B}=\{(1,2,0,1),(0,1,1,3),(2,0,1,1),(1,1,1,1)\}$ es una base de $\R^{4}$. Encuentra las coordenadas
        de $(a,b,c,d)\in\R^{4}$ en la base ordenada $\mathcal{B}$.
    \end{mdframed}

    \begin{proof}
        Sea
        \begin{equation*}
            \BB=\{(1,2,0,1),(0,1,1,3),(2,0,1,1),(1,1,1,1)\}=\{v_{1},v_{2},v_{3},v_{4}\}.
        \end{equation*}
        Supongamos
        \begin{equation*}
            \alpha_{1}v_{1}+\alpha_{2}v_{2}+\alpha_{3}v_{3}+\alpha_{4}v_{4}=0.
        \end{equation*}
        Esto da el sistema
        \begin{equation*}
            \begin{cases}
                \alpha_{1}+2\alpha_{3}+\alpha_{4}=0\\
                2\alpha_{1}+\alpha_{2}+\alpha_{4}=0\\
                \alpha_{2}+\alpha_{3}+\alpha_{4}=0\\
                \alpha_{1}+3\alpha_{2}+\alpha_{3}+\alpha_{4}=0
            \end{cases}
        \end{equation*}
        De la tercera ecuación
        \begin{equation*}
            \a_{2}=-\a_{3}-\a_{4}.
        \end{equation*}
        Sustituyendo en la segunda
        \begin{equation*}
            2\a_{1}-\a_{3}=0\implies \a_{3}=2\a_{1}.
        \end{equation*}
        Entonces
        \begin{equation*}
            \a_{2}=-2\a_{1}-\a_{4}.
        \end{equation*}
        Sustituimos en la cuarta
        \begin{equation*}
            \a_{1}+3(-2\a_{1}-\a_{4})+2\a_{1}+\a_{4}=0
        \end{equation*}
        \begin{equation*}
            -3\a_{1}-2\a_{4}=0.
        \end{equation*}
        De la primera
        \begin{equation*}
            \a_{1}+4\a_{1}+\a_{4}=0\implies 5\a_{1}+\a_{4}=0.
        \end{equation*}
        Así, el sistema resultante es
        \begin{equation*}
            \begin{cases}
                -3\a_{1}-2\a_{4}=0\\
                5\a_{1}+\a_{4}=0
            \end{cases}
        \end{equation*}
        Resolviendo
        \begin{equation*}
            \a_{1}=0\implies \a_{4}=0\implies \a_{3}=0\implies \a_{2}=0.
        \end{equation*}
        Por lo tanto, la única solución es la trivial, luego los vectores son linealmente
        independientes. Así, $\BB$ es una base de $\R^{4}$.\\
        Sea $x=(x_{1},x_{2},x_{3},x_{4})\in\R^{4}$. Buscamos
        \begin{equation*}
            x=\a_{1}v_{1}+\a_{2}v_{2}+\a_{3}v_{3}+a_{4}v_{4}.
        \end{equation*}
        Por lo tanto el sistema es
        \begin{equation*}
            \begin{cases}
                \a_{1}+2\a_{3}+\a_{4}=x_{1}\\
                2\a_{1}+\a_{2}+\a_{4}=x_{2}\\
                \a_{2}+\a_{3}+\a_{4}=x_{3}\\
                \a_{1}+3\a_{2}+\a_{3}+\a_{4}=x_{4}
            \end{cases}
        \end{equation*}
        Resolviendo de forma análoga
        \begin{equation*}
            \a_{3}=2\a_{1}+x_{3}-x_{2}
        \end{equation*}
        \begin{equation*}
            \a_{4}=x_{1}-5\a_{1}-2x_{3}+2x_{2}
        \end{equation*}
        Sustituyendo en la cuarta ecuación se obtiene
        \begin{equation*}
            \a_{1}=\frac{x_{1}+x_{2}-2x_{3}}{4}.
        \end{equation*}
        Luego,
        \begin{equation*}
            \a_{3}=\frac{x_{1}-x_{2}+2x_{3}}{2},\quad\a_{4}=\frac{-x_{1}-x_{2}+2x_{3}}{4},
        \end{equation*}
        \begin{equation*}
            \a_{2}=x_{3}-\a_{3}-\a_{4}=\frac{-x_{1}-3x_{2}+2x_{3}+2x_{4}}{4}.
        \end{equation*}
        En consecuencia
        \begin{equation*}
            [x]_{\BB}=\paren{\frac{x_{1}+x_{2}-2x_{3}}{4},\frac{-x_{1}-3x_{2}+2x_{3}+2x_{4}}{4},\frac{x_{1}-x_{2}+2x_{3}}{2},\frac{-x_{1}-x_{2}+2x_{3}}{4}}.
        \end{equation*}
    \end{proof}

    \begin{mdframed}[style=mdbluebox,frametitle={Ejercicio 3}]
        Considera $V_{2}$ el espacio vectorial sobre $\R$ de polinomios de grado menor o igual a 2 con coeficientes en $\R$. Sea
        $c\in\R\setminus\{0\}$, entonces $\BB_{2}=\{1,x+c,x^{2}+c^{2}\}$ es una base ordenada para $W_{2}$. Encuentra la matriz $P$
        de cambio de base de $\BB_{2}$ a la base ordenada $\BB_{1}=\{1,x,x^{2}\}$.
    \end{mdframed}

    \begin{solucion}
        La matriz de cambio de base $P$ de $\BB_{2}$ a $\BB_{1}$ está dada por
        \begin{equation*}
            P=[[v_{1}]_{\BB_{1}}\quad[v_{2}]_{\BB_{1}}\quad[v_{3}]_{\BB_{1}}]
        \end{equation*}
        donde $v_{1},v_{2},v_{3}$ son los vectores de $\BB_{2}$. Primer vector
        \begin{equation*}
            v_{1}=1=1\cdot1+0\cdot x+0\cdot x^{2}
        \end{equation*}
        \begin{equation*}
            [1]_{\BB_{1}}=
            \begin{pmatrix}
                1\\
                0\\
                0
            \end{pmatrix}.
        \end{equation*}
        Segundo vector
        \begin{equation*}
            v_{2}=x+c=c\cdot1+1\cdot x+0\cdot x^{2}
        \end{equation*}
        \begin{equation*}
            [x+c]_{\BB_{1}}=
            \begin{pmatrix}
                c\\
                1\\
                0
            \end{pmatrix}.
        \end{equation*}
        Tercer vector
        \begin{equation*}
            v_{3}=x^{2}+c^{2}=c^{2}\cdot1+0\cdot x+1\cdot x^{2}
        \end{equation*}
        \begin{equation*}
            [x^{2}+c^{2}]_{\BB_{1}}=
            \begin{pmatrix}
                c^{2}\\
                0\\
                1
            \end{pmatrix}.
        \end{equation*}
        Colocando los tres vectores como columnas
        \begin{equation*}
            P=
            \begin{pmatrix}
                1 & c & c^{2}\\
                0 & 1 & 0\\
                0 & 0 & 1
            \end{pmatrix}.
        \end{equation*}
    \end{solucion}
