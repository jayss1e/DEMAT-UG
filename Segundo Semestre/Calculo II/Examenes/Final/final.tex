    \begin{enumerate}
        \item Definamos
        \begin{equation*}
            f(x)=
            \begin{cases}
                x,\text{ si }x\in[1,2]\cap\mathbb{Q},\\
                0,\text{ si }x\in[1,2]\cap\mathbb{I}.
            \end{cases}
        \end{equation*}
        (a) Dibuja la grafica de $f$.\\
        (b) Determina si $f$ es integrable en $[1,2]$. Argumenta tu respuesta.\\
        (c) Si $f$ es integrable, calcula $\int_{1}^{2}f$.
        \item Calcula las integrales siguientes\\
        (a) $\int\cot^{2}(x)\csc(x)dx$.\\
        (b) $\int\frac{1}{x(\ln^{2}(x)+1)}dx$.\\
        (c) $\int\frac{1}{x\sqrt{4-x}}dx$.
        \item Calcula:
        \begin{equation*}
            \int\frac{x^{4}-x^{2}+11}{x^{2}-3x}dx
        \end{equation*}
        \item Sea $A$ un subconjunto no vacio y acotadon de números reales. Prueba que existe una sucesión $(a_{n})$ de puntos de $A$
        tal que $a_{n}\to\sup(A)$.
        \item Prueba que $\lim_{n\to\infty}\sqrt[n]{a^{n}+b^{n}}=\max\{a,b\}$ si $a$ y $b$ son números positivos.
        \item Considera una sucesión $(a_{n})$. Prueba que si $a_{2n}\to L$ y $a_{2n-1}\to L$ entonces $a_{n}\to L$.
        \item Determina la convergencia de las series siguientes
        \begin{multicols}{2}
            \begin{enumerate}
                \item[(a)] $\sum_{n=1}^{\infty}\frac{\sin(n)}{n^{2}+2}$
                \item[(b)] $\sum_{n=1}^{\infty}\frac{\cos(n\pi)}{n\pi}$ 
                \item[(c)] $\sum_{n=0}^{\infty}\frac{2^{n}}{n!}$
                \item[(d)] $\sum_{n=2}^{\infty}\frac{n}{\ln(n^{n})}$  
            \end{enumerate}
        \end{multicols}
        \item Sea $f_{n}(x)=x^{n}/(1+x^{n})$, $n\in\mathbb{N}$. Determina la convergencia puntual y uniforme de la sucesión de funciones
        $(f_{n})$ en el intervalo $[0,1]$.
    \end{enumerate}
