    \begin{enumerate}
        \item Calcula el área entre las gráficas de las funciones $f(x)=\sin(x)$ y $g(x)=x$ sobre
        el intervalo $[-\pi,\pi]$.
        \item \textbf{(Identidades suma-producto)} Prueba lo siguiente:\\
        (a) $\sin(x)+\sin(y)=2\sin(\frac{x+y}{2})\cos(\frac{x-y}{2})$.\\
        (b) $\cos(x)+\cos(y)=2\cos(\frac{x+y}{2})\cos(\frac{x-y}{2})$.
        \item Calcula
        \begin{equation*}
            \lim_{n\to\infty}\frac{\cos(n)}{n^{2}+1}.
        \end{equation*}
        \textbf{Sugerencia:} Ten presente que $\abs{\cos(x)}\leq1$ para toda $x$.
        \item Prueba que $\lim_{n\to\infty}\sqrt[n]{n^{2}+n}=1$. \textbf{Sugerencia:} Trata de
        usar el lema del sándwich.
        \item Considera una sucesión $(a_{n})$. Prueba que si $a_{n}\to L$ y $(a_{n_{k}})$ es
        una subsucesión de $(a_{n})$, entonces $a_{n_{k}}\to L$.
    \end{enumerate}
