    \textbf{Instrucciones:} Escribe limpia y ordenadamente el procedimiento
    (si lo hay) de cada ejercicio y no escribas las respuestas en la hoja
    del examen. ¡Suerte! 
    \begin{enumerate}
        \item Definamos
        \begin{equation*}
            f(x)=
            \begin{cases*}
                x,\text{ si }x\in[1,2]\cap\mathbb{Q},\\
                0,\text{ si }x\in[1,2]\cap\mathbb{I}.
            \end{cases*}
        \end{equation*}
        \begin{enumerate}
            \item[(a)] Dibuja la grafica de $f$.
            \item[(b)] Determina si $f$ es integrable en $[1,2]$. Argumenta tu respuesta.
            \item[(c)] Si $f$ es integrable en $[1,2]$. Argumenta tu respuesta. 
        \end{enumerate}
        \item Sea $f$ una función acotada en un intervalo $[a,b]$. Prueba que si $f$
        es continua en $[a,b]$, excepto en un número finito de puntos $p_{1},\dots,p_{n}\in[a,b]$,
        entonces $f$ es integrable en $[a,b]$.
        \item Muestra una funcion $f$ que no posea antiderivada (argumenta el porque no tiene
        antiderivada).
        \item Calcula las siguientes integrales
        \begin{enumerate}
            \item[(a)] $\displaystyle\int\left(\frac{9}{x^{2}+1}-x^{\frac{3}{4}}\right)dx$
            \item[(b)] $\displaystyle\int x\sec^{2}(x)dx$
            \item[(c)] $\displaystyle\int\frac{1}{x\ln(x)}dx$
            \item[(d)] $\displaystyle\int\sqrt{7+6x-x^{2}}dx$ 
        \end{enumerate}
        \item Calcula la integral
        \begin{equation*}
            \int\frac{x^{2}+5x}{2x^{3}-3x^{2}+1}dx
        \end{equation*}
    \end{enumerate}
