    \begin{mdframed}[style=mdbluebox,frametitle={Ejercicio 1}]
        Sea $f$ una función continua y no negativa en un intervalo $[a,b]$. Prueba que
        si $\int_{a}^{b}f=0$, entonces $f=0$. \textbf{Sugerencia:} Muestra la contrapositiva.
        Supon que $f\neq0$. Como $f$ es no negativa, entonces existe un punto $p\in[a,b]$ tal
        que $f(p)\gt0$. Supón primero que $p\in(a,b)$. Recuerda que como $f$ es continua en $p$
        y $f(p)\gt0$, existe un invervalo $I\subset[a,b]$ que contiene a $p$ tal que $f(x)\gt0$
        para toda $x\in I$. Prueba que esto implica que $\int_{a}^{b}f\gt0$. Para los casos en que
        $p=a$ o $p=b$ se tiene una sugerencia similar.
    \end{mdframed}

    \begin{proof}
        Supongamos que $f\neq0$. Entonces existe $p\in[a,b]$ tal que
        $f(p)\gt0$. Como $f$ es continua en $p$ y $f(p)\gt0$, por la definición
        de continuidad existe $\delta\gt0$ tal que
        \begin{equation*}
            \abs{x-p}\lt\delta\implies f(x)\gt\frac{f(p)}{2}\gt0.
        \end{equation*}
        Sea $I=[c,d]\subset[a,b]$. Existe $p\in[a,b]$ tal que
        \begin{equation*}
            f(p)\gt0.
        \end{equation*}
        Por continuidad de $f$ en $p$, existe $\delta\gt0$ tal que
        \begin{equation*}
            \abs{x-p}\lt\delta\implies f(x)\gt\frac{f(p)}{2}.
        \end{equation*}
        Definamos
        \begin{equation*}
            c=p-\delta,\qquad d=p+\delta,
        \end{equation*}
        Entonces
        \begin{equation*}
            f(x)\gt\frac{f(p)}{2}\quad\forall x\in[c,d].
        \end{equation*}
        Por lo tanto
        \begin{equation*}
            \int_{a}^{b}f(x)dx\geq\int_{c}^{d}f(x)dx\gt\int_{c}^{d}\frac{f(p)}{2}dx=\frac{f(p)}{2}(d-c).
        \end{equation*}
        Como $f(p)\gt0$ y $d-c\gt0$,
        \begin{equation*}
            \int_{a}^{b}f(x)dx\gt0.
        \end{equation*}
        Por contrapositiva concluimos que
        \begin{equation*}
            f=0.
        \end{equation*}
    \end{proof}

    \begin{mdframed}[style=mdbluebox,frametitle={Ejercicio 2}]
        Considera las funciones $f:[1,\infty)\to\mathbb{R}$, $g:[1,\infty)\to\mathbb{R}$
        y $h:[1,\infty)\to\mathbb{R}$ dadas por
        \begin{equation*}
            f(x)=\frac{1}{\sqrt{x}}, g(x)=\frac{1}{x}\text{ y } h(x)=\frac{1}{x^{2}}.
        \end{equation*}
        \begin{enumerate}
            \item[(a)] Dibuja las gráfica de $f$, $g$ y $h$.
            \item[(b)] Calcula el área bajo la grafica de cada una de las funciones $f,$ $g$ y $h$.
        \end{enumerate}
    \end{mdframed}

    \begin{solucion}
        \begin{enumerate}
            \item[(a)] Dibuja las graficas de $f,g$ y $h$\\
            \begin{figure}[h]
                \centering
                \includegraphics[width=0.32\textwidth]{f.pdf}
                \includegraphics[width=0.32\textwidth]{g.pdf}
                \includegraphics[width=0.32\textwidth]{h.pdf}
            \end{figure}
            \item[(b)] Calcula el área bajo la grafica de cada una de las funciones $f,$ $g$ y $h$.
            \begin{enumerate}
                \item[1.] $f(x)=\frac{1}{\sqrt{x}}$.\\
                El area bajo la grafica de $f$ en el intervalo $[1,\infty)$ se describe mediante la integral impropia
                \begin{equation*}
                    \int_{1}^{\infty}\frac{1}{\sqrt{x}}dx,
                \end{equation*}
                Por definición de integral impropio, se define
                \begin{equation*}
                    \int_{1}^{\infty}\frac{1}{\sqrt{x}}dx:=\lim_{t\to\infty^{+}}\int_{1}^{t}\frac{1}{\sqrt{x}}dx.
                \end{equation*}
                Calculamos primero la integral en $[1,t]$
                \begin{align*}
                    \int_{1}^{t}\frac{1}{\sqrt{x}}dx&=\int_{1}^{t}x^{-\frac{1}{2}}dx\\
                    &=\big[2x^{\frac{1}{2}}\big]_{1}^{t}\\
                    &=2t^{\frac{1}{2}}-2
                \end{align*}
                Tomando el limite cuando $t\to\infty^{+}$
                \begin{equation*}
                    \int_{1}^{t}\frac{1}{\sqrt{x}}dx=\lim_{t\to\infty^{+}}2t^{\frac{1}{2}}-2=\infty.
                \end{equation*}
                Así, el area bajo la grafica diverge.
                \item[2.] $g(x)=\frac{1}{x}$.\\
                El area bajo la grafica de $f$ en el intervalo $[1,\infty)$ se describe mediante la integral impropia
                \begin{equation*}
                    \int_{1}^{\infty}\frac{1}{x}dx,
                \end{equation*}
                Por definición de integral impropio, se define
                \begin{equation*}
                    \int_{1}^{\infty}\frac{1}{x}dx:=\lim_{t\to\infty^{+}}\int_{1}^{t}\frac{1}{x}dx.
                \end{equation*}
                Calculamos primero la integral en $[1,t]$
                \begin{align*}
                    \int_{1}^{t}\frac{1}{x}dx&=\int_{1}^{t}x^{-1}dx\\
                    &=\big[\ln(x)\big]_{1}^{t}\\
                    &=\ln(t)-\ln(1)\\
                    &=\ln(t).
                \end{align*}
                Tomando el limite cuando $t\to\infty^{+}$
                \begin{equation*}
                    \int_{1}^{t}\frac{1}{x}dx=\lim_{t\to\infty^{+}}\ln(t)=\infty.
                \end{equation*}
                Así, el area bajo la grafica diverge.
                \item[3.] $h(x)=\frac{1}{x^{2}}$.\\
                El area bajo la grafica de $f$ en el intervalo $[1,\infty)$ se describe mediante la integral impropia
                \begin{equation*}
                    \int_{1}^{\infty}\frac{1}{x^{2}}dx,
                \end{equation*}
                Por definición de integral impropio, se define
                \begin{equation*}
                    \int_{1}^{\infty}\frac{1}{x^{2}}dx:=\lim_{t\to\infty^{+}}\int_{1}^{t}\frac{1}{x^{2}}dx.
                \end{equation*}
                Calculamos primero la integral en $[1,t]$
                \begin{align*}
                    \int_{1}^{t}\frac{1}{x^{2}}dx&=\int_{1}^{t}x^{-2}dx\\
                    &=\big[-\frac{1}{x}\big]_{1}^{t}\\
                    &=-\frac{1}{t}+1
                \end{align*}
                Tomando el limite cuando $t\to\infty^{+}$
                \begin{equation*}
                    \int_{1}^{t}\frac{1}{x}dx=\lim_{t\to\infty^{+}}-\frac{1}{t}+1=1.
                \end{equation*}
                Así, el area bajo la grafica es 1.
            \end{enumerate}
        \end{enumerate}
    \end{solucion}

    \begin{mdframed}[style=mdbluebox,frametitle={Ejercicio 3}]
        Muestra que el área del círculo con centro en el origen y radio $r$ es $\pi r^{2}$.
    \end{mdframed}

    \begin{proof}
        Sea $C$ el circulo de radio $r\gt0$ con centro en el origen. Su ecuación es
        $x^{2}+y^{2}=r^{2}$. La semicircunferencia superior puede describirse como la
        grafica de la función $f:[-r,r]\to\mathbb{R}$ definida por
        \begin{equation*}
            f(x)=\sqrt{r^{2}-x^{2}}.
        \end{equation*}
        Así, el area de la semicircunferencia esta dada por
        \begin{equation*}
            A_{semi}=\int_{-r}^{r}\sqrt{r^{2}-x^{2}}dx.
        \end{equation*}
        Tomemos $x=r\sin(\theta)$ y $dx=r\cos(\theta)d\theta$. En consecuencia,
        \begin{align*}
            \int_{r}^{r}\sqrt{r^{2}-x^{2}}dx&=\int_{-r}^{r}\sqrt{r^{2}-r^{2}\sin^{2}(\theta)}r\cos(\theta)d\theta\\
            &=r^{2}\int_{-r}^{r}\cos^{2}(\theta)d\theta\\
            &=r^{2}\int_{-r}^{r}\frac{1+\cos(2\theta)}{2}d\theta\\
            &=r^{2}\big[\int_{-r}^{r}\frac{1}{2}d\theta+\int_{-r}^{r}\frac{\cos(2\theta)}{2}\big]d\theta\\
            &=\big[\frac{r^{2}\theta+r^{2}\sin(\theta)\cos(\theta)}{2}\big]
        \end{align*}
        De $x=r\sin(\theta)$ tenemos que $\theta=\arcsin(\frac{x}{r})$, $r\sin(\theta)=x$ y que $r\cos(\theta)=\sqrt{r^{2}-x^{2}}$. Así,
        \begin{align*}
            \big[\frac{r^{2}\theta+r^{2}\sin(\theta)\cos(\theta)}{2}\big]_{-r}^{r}=\big[\frac{r^{2}\arcsin(\frac{x}{r})+x\sqrt{r^{2}-x^{2}}}{2}\big]_{-r}^{r}
        \end{align*}
        Evaluando en los limites de integración
        \begin{align*}
            \big[\frac{r^{2}\arcsin(\frac{x}{r})+x\sqrt{r^{2}-x^{2}}}{2}\big]_{-r}^{r}&=\big[\frac{r^{2}\arcsin(\frac{r}{r})+r\sqrt{r^{2}-r^{2}}}{2}\big]-\big[\frac{r^{2}\arcsin(-\frac{r}{r})-r\sqrt{r^{2}-r^{2}}}{2}\big]\\
            &=\big[\frac{r^{2}\frac{\pi}{2}}{2}\big]-\big[\frac{-r^{2}\frac{\pi}{2}}{2}\big]\\
            &=\frac{r^{2}\pi}{4}+\frac{r^{2}\pi}{4}\\
            &=\frac{r^{2}\pi}{2}.
        \end{align*}
        De esta forma el area de la semicircunferencia es $\frac{r^{2}\pi}{2}$ pero
        \begin{equation*}
            A=2A_{semi}=2\frac{r^{2}\pi}{2}.
        \end{equation*}
        Así, el area de la circunferencia con radio $r$ es
        \begin{equation*}
            A=\pi r^{2}.
        \end{equation*}
    \end{proof}

    \begin{mdframed}[style=mdbluebox,frametitle={Ejercicio 4}]
        Utiliza la circunferencia trigonometrica para calcular, si es que existe, el valor de las funciones trigonometricas del angulo
        $\theta$ de cada uno de los incisos siguientes (haz un dibujo).
        \begin{multicols}{2}
            \begin{enumerate}
                \item[(a)]  $\theta=\pi/4.$
                \item[(b)]  $\theta=3\pi/4.$
                \item[(c)]  $\theta=5\pi/4.$
                \item[(d)]  $\theta=7\pi/4.$
            \end{enumerate}
        \end{multicols}
        \textbf{Sugerencia:} Resuelve el primer inciso y usa la simetría en los demás.
    \end{mdframed}

    \begin{solucion}
        \begin{enumerate}
            \item[(a)] $\theta=\pi/4$\\
            El angulo forma un triangulo isoceles, de esta forma
            \begin{equation*}
                \sin^{2}(\theta)+\cos^{2}(\theta)=1.
            \end{equation*}
            Si $\sin(\theta)=\cos^{2}(\theta)=a$, entonces
            \begin{equation*}
                2a^{2}=1.
            \end{equation*}
            Luego,
            \begin{equation*}
                a=\frac{\sqrt{2}}{2}.
            \end{equation*}
            En consecuencia
            \begin{equation*}
                \sin(\frac{\pi}{4})=\frac{\sqrt{2}}{2},\qquad \cos(\frac{\pi}{4})=\frac{\sqrt{2}}{2}
            \end{equation*}
            Entonces,
            \begin{equation*}
                \tan(\frac{\pi}{4})=1,\qquad \csc(\frac{\pi}{4})=\sqrt{2},\qquad \sec(\frac{\pi}{4})=\sqrt{2},\qquad \cot(\frac{\pi}{4})=1.
            \end{equation*}
            \begin{center}
                \includegraphics[width=0.35\textwidth]{a.pdf}
            \end{center}
            \item[(b)] $\theta=3\pi/4$.\\
            Notemos que
            \begin{equation*}
                \frac{3\pi}{4}=\pi-\frac{\pi}{4}.
            \end{equation*}
            Entonces,
            \begin{equation*}
                 \sin(\frac{3\pi}{4})=\sin(\pi-\frac{\pi}{4})\text{ y }\cos(\frac{3\pi}{4})=\cos(\pi-\frac{\pi}{4}).
            \end{equation*}
            y por lo prosicion 31 se sigue que
            \begin{align*}
                \sin(\pi-\frac{\pi}{4})=\sin(\pi)\cos(-\frac{\pi}{4})+\sin(-\frac{\pi}{4})\cos(\pi).\\
                \cos(\pi-\frac{\pi}{4})=\cos(\pi)\cos(-\frac{\pi}{4})-\sin(\pi)\sin(-\frac{\pi}{4}).
            \end{align*}
            Luego, por el inciso anterior y la tarea anterior
            \begin{align*}
                \sin(\frac{3\pi}{4})=-\frac{\sqrt{2}}{2}\\
                \cos(\frac{3\pi}{4})=\frac{\sqrt{2}}{2}.
            \end{align*}
            De aquí se sigue que
            \begin{equation*}
                \tan(\frac{3\pi}{4})=-1,\qquad\csc(\frac{3\pi}{4})=-\sqrt{2},\qquad\sec(\frac{3\pi}{4})=\sqrt{2},\qquad\cot(\frac{3\pi}{4})=-1
            \end{equation*}
            \begin{center}
                \includegraphics[width=0.35\textwidth]{b.pdf}
            \end{center}
            \item[(c)] $\theta=5\pi/4$.\\
            Notemos que 
            \begin{equation*}
                \frac{5\pi}{4}=\pi+\frac{\pi}{4}
            \end{equation*}
            Este angulo esta en el tercer cuadrante y aqui el seno es negativo y el coseno tambien, por lo tanto
            \begin{equation*}
                \sin(\frac{5\pi}{4})=-\frac{\sqrt{2}}{2},\qquad\cos(\frac{5\pi}{4})=-\frac{\sqrt{2}}{2}
            \end{equation*}
            En consecuencia
            \begin{equation*}
                \tan(\frac{5\pi}{4})=1,\qquad \csc(\frac{5\pi}{4})=-1,\qquad\sec(\frac{5\pi}{4})=-1,\qquad\cot(\frac{5\pi}{4})=1.

            \begin{center}
                \includegraphics[width=0.35\textwidth]{c.pdf}
            \end{center}
            \item[(d)] $\theta=7\pi/4$.\\
            Notemos que
            \begin{equation*}
                \frac{7\pi}{4}=2\pi-\frac{\pi}{4}.
            \end{equation*}
            En este cuadrante el seno es negativo y el coseno positivo. por lo tanto
            \begin{equation*}
                \sin(\frac{7\pi}{4})=-\frac{\sqrt{2}}{2},\qquad\cos(\frac{7\pi}{4})=\frac{\sqrt{2}}{2}.
            \end{equation*}

            \begin{equation*}

    \begin{mdframed}[style=mdbluebox,frametitle={Ejercicio 5}]
        Prueba el corolario 1
    \end{mdframed}

    \begin{proof}
        Notemos que
        \begin{equation*}
            \tan(x+y)=\frac{\sin(x+y)}{\cos(x+y)}.
        \end{equation*}
        Luego, por la proposición 31 se sigue que
        \begin{align*}
            \tan(x+y)&=\frac{\sin(x+y)}{\cos(x+y)}\\
            &=\frac{\sin(x+y)}{\cos(x+y)}\frac{\cos(x)\cos(y)}{\cos(x)\cos(y)}\\
            &=\frac{\frac{\sin(x)\sin(y)+\sin(y)\sin(x)}{\cos(x)\cos(y)}}{\frac{\cos(x)\cos(y)-\sin(x)\sin(y)}{\cos(x)\cos(y)}}\\
            &=\frac{\frac{\sin(x)}{\cos(x)}+\frac{\sin(y)}{\cos(y)}}{1-\frac{\sin(x)}{\cos(x)}\frac{\sin(y)}{\cos(y)}}\\
            &=\frac{\tan(x)+\tan(y)}{1-\tan(x)\tan(y)}.
        \end{align*}
        Concluyendo que
        \begin{equation*}
            \tan(x+y)=\frac{\tan(x)+\tan(y)}{1-\tan(x)\tan(y)}.
        \end{equation*}
    \end{proof}

    \begin{mdframed}[style=mdbluebox,frametitle={Ejercicio 6}]
        Prueba el inciso (c) de la proposición 35. \textbf{Sugerencia:} Prueba el inciso (c)
        primero cuando $r$ es un número natural, luego cuando $r$ es un número entero y
        finalmente cuando $r$ es un número racional.
    \end{mdframed}

    \begin{proof}
        Recordemos que $\ln(x)$ es la funcion inversa de la exponencial $e^{x}$. En particular,
        \begin{equation*}
            \ln(e^{t})=t,\qquad e^{\ln(x)}=x
        \end{equation*}
        Adémas, para cualquier $a,b\in\mathbb{R}$,
        \begin{equation*}
            e^{a+b}=e^{a}e^{b}.
        \end{equation*}
        Como $x\gt0$,
        \begin{equation*}
            x=e^{\ln(x)}
        \end{equation*}
        Entonces,
        \begin{equation*}
            x^{r}=(e^{\ln(x)})^{r}.
        \end{equation*}
        Por la proposicion 38,
        \begin{equation*}
            x^{r}=e^{r\ln(x)}.
        \end{equation*}
        Tomando el logaritmo natural a ambos lados
        \begin{equation*}
            \ln(x^{r})=\ln(e^{r\ln(x)})
        \end{equation*}
        Como el $\ln$ es la inversa de $e^{x}$
        \begin{equation*}
            \ln(e^{r\ln(x)})=r\ln(x)
        \end{equation*}
        Así,
        \begin{equation*}
            \ln(x^{r})=r\ln(x)
        \end{equation*}
    \end{proof}

    \begin{mdframed}[style=mdbluebox,frametitle={Ejercicio 7}]
        \textbf{(La función $x^{p}$)} Sea $p$ un número real.
        \begin{enumerate}
            \item[(a)] Encuentra el dominio de la función $p$ (por supuesto, el dominio de la
            función $x^{p}$ depende de $p$).
            \item[(b)] Prueba que
            \begin{equation*}
                (x^{p})^{\prime}=px^{p-1}
            \end{equation*}
            (esto ultimo ya lo sabíamos si $p$ es un número racional, véase por ejemplo
            \textbf{[?]}).
        \end{enumerate}
    \end{mdframed}

    \begin{solucion}
        \begin{enumerate}
            \item[(a)] Encuentra el dominio de la función $p$ (por supuesto, el dominio de la
            función $x^{p}$ depende de $p$).\\
            Sea $p\in\mathbb{R}$. La función potencia para exponente real se
            puede definir mediante logaritmos, así
            \begin{equation*}
                x^{p}=e^{p\ln(x)}
            \end{equation*}
            Pero $\ln(x)$ solo esta definida para para $x\gt0$. Por lo tanto,
            para $p\in\mathbb{R}$ la función queda definida en $(0,\infty)$.
            \item[(b)] Prueba que $(x^{p})^{\prime}=px^{p-1}$
            Definamos
            \begin{equation*}
                x=e^{p\ln(x)},\qquad x\gt0.
            \end{equation*}
            Derivamos aplicando la regla de la cadena. Notemos que
            \begin{equation*}
                \frac{d}{dx}(e^{u})=e^{u}u^{\prime}\quad\text{ y }\quad \frac{d}{dx}\ln(x)=\frac{1}{x}.
            \end{equation*}
            Entonces
            \begin{equation*}
                \frac{d}{dx}(x^{p})=\frac{d}{dx}(e^{p\ln(x)}).
            \end{equation*}
            Aplicando la regla de la cadena
            \begin{equation*}
                \frac{d}{dx}(e^{p\ln(x)})=e^{p\ln(x)}\frac{p}{x}
            \end{equation*}
            Pero
            \begin{equation*}
                (e^{p\ln(x)})=x^{p}
            \end{equation*}
            Por lo que
            \begin{align*}
                \frac{d}{dx}(x^{p})&=x^{p}\frac{p}{x}\\
                &=px^{p-1}
            \end{align*}
        \end{enumerate}
    \end{solucion}

    \begin{mdframed}[style=mdbluebox,frametitle={Ejercicio 8}]
        Prueba que $a_{n}\to0$ ssi $\abs{a_{n}}\to0$.
    \end{mdframed}

    \begin{proof}
        Supongamos que $a_{n}\to0$. Por definición de límite de sucesiones
        \begin{equation*}
            \forall\varepsilon\gt0\quad\exists N\in\mathbb{N}
        \end{equation*}
        tal que si $n\geq N$, entonces
        \begin{equation*}
            \abs{a_{n}-0}\lt\varepsilon.
        \end{equation*}
        Pero
        \begin{equation*}
            \abs{a_{n}-0}=\abs{a_{n}}.
        \end{equation*}
        Por lo tanto
        \begin{equation*}
            \abs{a_{n}}\lt\varepsilon\text{ para todo }n\geq N.
        \end{equation*}
        Esto es precisamente
        \begin{equation*}
            \abs{a_{n}}\to0.
        \end{equation*}
        Supongamos ahora que $\abs{a_{n}}\to0$. Por definición de límite de sucesiones
        \begin{equation*}
            \forall\varepsilon\gt0\quad\exists N\in\mathbb{N}
        \end{equation*}
        tal que si $n\geq N$,
        \begin{equation*}
            \abs{\abs{a_{n}}-0}\lt\varepsilon.
        \end{equation*}
        Pero
        \begin{equation*}
            \abs{\abs{a_{n}}-0}=\abs{a_{n}}.
        \end{equation*}
        Así,
        \begin{equation*}
            \abs{a_{n}}\lt\varepsilon\text{ para todo }n\geq N.
        \end{equation*}
        Esto equivale a
        \begin{equation*}
            \abs{a_{n}-0}\lt\varepsilon
        \end{equation*}
        que es exactamente la definición de
        \begin{equation*}
            \abs{a_{n}}\to0.
        \end{equation*}
    \end{proof}
