\documentclass[tikz,border=5pt]{standalone}

\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

\begin{document}

\begin{tikzpicture}
\begin{axis}[
    view={120}{25},
    axis lines=center,
    xlabel={$x$},
    ylabel={$y$},
    zlabel={$z$},
    xmin=-1.3, xmax=1.3,
    ymin=-1.3, ymax=1.3,
    zmin=-4, zmax=4,
    axis equal,
    grid=major,
]

\addplot3[
    surf,
    domain=0:360,
    y domain=-4:4,
    samples=120,
    samples y=25,
    opacity=0.7,
]
(
    {cos(2*x)*cos(x)},
    {cos(2*x)*sin(x)},
    {y}
);

\end{axis}
\end{tikzpicture}

\end{document}
