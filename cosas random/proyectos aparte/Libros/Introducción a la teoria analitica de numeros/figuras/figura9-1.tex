\documentclass[border=4pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{arrows.meta,decorations.markings}

\begin{document}
\begin{tikzpicture}[>=Latex, scale=1.5]

    % Re^{i pi/4} = (\c,\c)
    \def\c{1.3}
    \coordinate (A)  at (-0.5-\c,-\c);
    \coordinate (A1) at (0.5-\c,-\c);
    \coordinate (B)  at (-0.5+\c,\c);
    \coordinate (B1) at (0.5+\c,\c);
    \coordinate (O)  at (0,0);

    % axes
    \draw (-2.4,0) -- (2.4,0);
    \draw (0,-1.9) -- (0,1.9);

    % oriented parallelogram: A -> A+1 -> B+1 -> B -> A
    \draw[thick,
        postaction={decorate,decoration={markings,
            mark=at position 0.5 with {\arrow{Latex}}}}] (A) -- (A1);
    \draw[thick,
        postaction={decorate,decoration={markings,
            mark=at position 0.5 with {\arrow{Latex}}}}] (A1) -- (B1);
    \draw[thick,
        postaction={decorate,decoration={markings,
            mark=at position 0.5 with {\arrow{Latex}}}}] (B1) -- (B);
    \draw[thick,
        postaction={decorate,decoration={markings,
            mark=at position 0.5 with {\arrow{Latex}}}}] (B) -- (A);

    % labels
    \node[below left]  at (A)  {\small $A$};
    \node[below right] at (A1) {\small $A+1$};
    \node[above left]  at (B)  {\small $B$};
    \node[above right] at (B1) {\small $B+1$};
    \node[below right] at (O)  {\small $0$};

\end{tikzpicture}
\end{document}
